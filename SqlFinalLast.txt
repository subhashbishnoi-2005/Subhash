-- q11
select book_id,count(*) from transactions group by book_id order by count(*) desc limit 1;

-- q12
select b.title,a.name from books b join authors a on b.author_id=a.author_id;

-- q13
select m.name,b.title from members m left join transactions t on m.member_id=t.member_id left join books b on t.book_id=b.book_id;

-- q14 
select b.title from books b left join transactions t on b.book_id=t.book_id where t.book_id is null;

-- q15
